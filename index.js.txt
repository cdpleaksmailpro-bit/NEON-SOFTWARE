  Client,
  GatewayIntentBits,
  EmbedBuilder,

  ButtonBuilder,
  ButtonStyle

// =====================================
// NEON SOFTWARE CONFIGURATION
// =====================================

  CLIENT_ID: "1570948368375615568",
  CLIENT_SECRET: process.env.CLIENT_SECRET,

  REDIRECT_URI: process.env.REDIRECT_URI,
  BASE_URL: process.env.BASE_URL,

  BOT_TOKEN: process.env.BOT_TOKEN,

  !CONFIG.REDIRECT_URI ||

});

const pendingStates = new Map();

// =====================================
// HTML SECURITY
// =====================================

function escapeHTML(value) {
  return String(value).replace(/[&<>"']/g, c => ({
    "&": "&amp;",
    "<": "&lt;",
    ">": "&gt;",
    '"': "&quot;",
    "'": "&#39;"
  })[c]);
}

// =====================================
// OAUTH LOGIN
// =====================================

app.get("/login", (req, res) => {
  const state = crypto.randomBytes(32).toString("hex");

  pendingStates.set(state, Date.now());

  const params = new URLSearchParams({
    client_id: CONFIG.CLIENT_ID,
    redirect_uri: CONFIG.REDIRECT_URI,
    response_type: "code",
    scope: "identify",
    state
  });

  res.redirect(
    "https://discord.com/oauth2/authorize?" + params
  );
});

// =====================================
// DISCORD CALLBACK
// =====================================

app.get("/callback", async (req, res) => {
  try {
    const { code, state, error } = req.query;

    if (error) {
      return res.status(400).send(page("Cancelled", `
        <h2>Verification cancelled</h2>
        <p>You cancelled Discord authorization.</p>
        <button onclick="location.href='/'">Try again</button>
      `));
    }

    if (
      typeof code !== "string" ||
      typeof state !== "string" ||
      !pendingStates.has(state) ||
      Date.now() - pendingStates.get(state) > 300000
    ) {
      if (typeof state === "string") {
        pendingStates.delete(state);
      }

      return res.status(400).send(page("Error", `
        <h2>Invalid request</h2>
        <p>Your verification request expired or is invalid.</p>
        <button onclick="location.href='/'">Try again</button>
      `));
    }

    pendingStates.delete(state);

    // EXCHANGE OAUTH CODE

    const tokenResponse = await fetch(
      "https://discord.com/api/v10/oauth2/token",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/x-www-form-urlencoded"
        },
        body: new URLSearchParams({
          client_id: CONFIG.CLIENT_ID,
          client_secret: CONFIG.CLIENT_SECRET,
          grant_type: "authorization_code",
          code,
          redirect_uri: CONFIG.REDIRECT_URI
        })
      }
    );

    const tokens = await tokenResponse.json();

    if (!tokenResponse.ok || !tokens.access_token) {
      console.error("OAuth error:", tokens);
      throw new Error("OAuth authentication failed");
    }

    // GET USER

    const userResponse = await fetch(
      "https://discord.com/api/v10/users/@me",
      {
        headers: {
          Authorization: "Bearer " + tokens.access_token
        }
      }
    );

    const user = await userResponse.json();

    if (!userResponse.ok || !user.id) {
      throw new Error("Unable to retrieve Discord user");
    }

    // FETCH GUILD

    const guild = await bot.guilds.fetch(CONFIG.GUILD_ID);

    let member;

    try {
      member = await guild.members.fetch(user.id);
    } catch {
      return res.status(403).send(page("Not a member", `
        <h2>Server membership required</h2>
        <p>
          Join the NEON SOFTWARE Discord server
          before verifying your account.
        </p>
        <button onclick="location.href='/'">Try again</button>
      `));
    }

    // ASSIGN ROLE

    if (!member.roles.cache.has(CONFIG.ROLE_ID)) {
      await member.roles.add(
        CONFIG.ROLE_ID,
        "NEON SOFTWARE account verification"
      );
    }

    console.log("VERIFIED USER:", user.username);

    res.send(page("Success", `
      <div class="success">✓</div>
      <h2 class="success">Success!</h2>
      <p>
        Welcome, ${escapeHTML(user.username)}.<br>
        Your Discord account has been verified successfully.
      </p>
      <p>You can now return to Discord.</p>
    `));

  } catch (error) {
    console.error("Verification error:", error);

    res.status(500).send(page("Error", `
      <h2>Verification failed</h2>
      <p>Please try again or contact the server administration.</p>
      <button onclick="location.href='/'">Try again</button>
    `));
  }
});

// =====================================
// CLEAN EXPIRED STATES
// =====================================

setInterval(() => {
  const now = Date.now();

  for (const [state, timestamp] of pendingStates.entries()) {
    if (now - timestamp > 300000) {
      pendingStates.delete(state);
    }
  }
}, 60000);

// =====================================
// DISCORD PANEL
// =====================================

bot.once("clientReady", async () => {
  console.log("NEON SOFTWARE BOT CONNECTED");

  try {
    const channel = await bot.channels.fetch(
      CONFIG.VERIFY_CHANNEL_ID
    );

    if (!channel || !channel.isTextBased() || !channel.messages) {
      throw new Error("Verification channel not found.");
    }

    const embed = new EmbedBuilder()
      .setColor("#171717")
      .setTitle("NEON SOFTWARE")
      .setDescription(
        "Welcome to **NEON SOFTWARE**!\n\n" +
        "To gain access to the server, you must first verify your Discord account.\n\n" +
        "Click the button below to begin the verification process.\n\n" +
        "🔒 Secure verification\n" +
        "⚡ Automatic role assignment"
      )
      .setFooter({
        text: "NEON SOFTWARE • Verification"
      });

    const row = new ActionRowBuilder().addComponents(
      new ButtonBuilder()
        .setLabel("Verify with Discord")
        .setStyle(ButtonStyle.Link)
        .setURL(CONFIG.BASE_URL + "/login")
    );

    const messages = await channel.messages.fetch({
      limit: 50
    });

    const existing = messages.find(message =>
      message.author.id === bot.user.id &&
      message.embeds.some(embed =>
        embed.title === "NEON SOFTWARE"
      )
    );

    if (existing) {
      await existing.edit({
        embeds: [embed],
        components: [row]
      });
    } else {
      await channel.send({
        embeds: [embed],
        components: [row]
      });
    }

    console.log("VERIFICATION PANEL READY");

  } catch (error) {
    console.error("Panel error:", error);
  }
});
